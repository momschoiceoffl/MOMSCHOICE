    document.addEventListener("DOMContentLoaded", () => {
      const params = new URLSearchParams(window.location.search);
      
      const pId = params.get("id") || "Unknown ID";
      const pTitle = params.get("title") || "Unknown Product";
      const pPrice = Number(params.get("price")) || 0;
      const qty = Number(params.get("qty")) || 1;
      
      const cName = params.get("cName") || "Walk-in Customer";
      const cPhone = params.get("cPhone") || "-";
      const cAddr = params.get("cAddr") || "-";
      
      const orderId = params.get("orderId") || ("ORD-" + Math.floor(1000 + Math.random() * 9000));

      document.getElementById("lblInvNum").innerText = "INV-" + Math.floor(1000 + Math.random() * 9000);
      document.getElementById("lblDate").innerText = new Date().toLocaleDateString();
      document.getElementById("lblQr").src = "https://quickchart.io/qr?text=" + encodeURIComponent(orderId) + "&size=250&margin=1";

      document.getElementById("lblCustName").innerText = cName;
      document.getElementById("lblCustAddr").innerHTML = cAddr.replace(/,/g, "<br>");
      document.getElementById("lblCustPhone").innerText = cPhone;

      const itemTotal = pPrice * qty;
      document.getElementById("lblProdId").innerText = pId;
      document.getElementById("lblProdTitle").innerText = pTitle;
      document.getElementById("lblQty").innerText = qty;
      document.getElementById("lblRate").innerText = "₹" + pPrice;
      document.getElementById("lblAmount").innerText = "₹" + itemTotal;

      const shipping = itemTotal > 2000 ? 0 : 60;
      const grandTotal = itemTotal + shipping;
      document.getElementById("lblSubtotal").innerText = "₹" + itemTotal;
      document.getElementById("lblShipping").innerText = shipping === 0 ? "Free" : "₹" + shipping;
      document.getElementById("lblTotal").innerText = "₹" + grandTotal;
    });
  </script>
