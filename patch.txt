    async function generateInvoiceFromForm() {
      const query = document.getElementById("invSearchInput").value.trim().toLowerCase();
      const custName = document.getElementById("invCustName").value.trim();
      const custPhone = document.getElementById("invCustPhone").value.trim();
      const custAddress = document.getElementById("invCustAddress").value.trim();
      const qty = document.getElementById("invQty").value;
      const shipping = document.getElementById("invShipping").value;

      if (!query) return alert("Please scan or enter a Product ID first.");
      if (!custName || !custPhone || !custAddress) return alert("Please fill in all customer details.");

      const product = productsList.find(p => p.id.toLowerCase() === query || p.title.toLowerCase().includes(query));
      if (!product) return alert("❌ Product not found in Database.");

      // 1. Generate Sequential Invoice Number (Prefix-YYMMDD-XXX)
      const d = new Date();
      const yy = d.getFullYear().toString().slice(-2);
      const mm = String(d.getMonth() + 1).padStart(2, '0');
      const dd = String(d.getDate()).padStart(2, '0');
      
      // Create a prefix based on the category (e.g., SAR for Sarees)
      const catPrefix = (product.category || "GEN").substring(0, 3).toUpperCase();

      // Calculate sequential order count for today
      let todayCount = 1;
      try {
        const logRes = await fetch("http://localhost:5000/get-invoice-count?prefix=" + catPrefix + "&date=" + yy + mm + dd);
        if (logRes.ok) {
          const logData = await logRes.json();
          todayCount = logData.count + 1; // Increment for the new invoice
        }
      } catch (e) {
        console.log("Could not fetch invoice count, defaulting to 001");
      }

      const seq = String(todayCount).padStart(3, '0');
      const invNumber = `${catPrefix}-${yy}${mm}${dd}-${seq}`;

      // 2. Save the Invoice to Google Sheets
      const invoiceData = {
        action: "save_invoice",
        invNumber: invNumber,
        orderId: "Auto",
        cName: custName,
        cPhone: custPhone,
        cAddr: custAddress,
        id: product.id,
        title: product.title,
        qty: qty,
        price: product.price,
        grandTotal: (Number(product.price) * Number(qty)) + Number(shipping)
      };

      // Send it to the server silently in the background
      fetch("http://localhost:5000/save", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(invoiceData)
      });

      // 3. Open label.html and pass the new Invoice Number
      const url = `label.html?id=${encodeURIComponent(product.id)}` +
                  `&title=${encodeURIComponent(product.title)}` +
                  `&price=${product.price}` +
                  `&qty=${qty}` +
                  `&shipping=${shipping}` +
                  `&invNum=${encodeURIComponent(invNumber)}` +
                  `&cName=${encodeURIComponent(custName)}` +
                  `&cPhone=${encodeURIComponent(custPhone)}` +
                  `&cAddr=${encodeURIComponent(custAddress)}`;
      
      window.open(url, '_blank');
      
      // Clear the form
      document.getElementById("invSearchInput").value = "";
      document.getElementById("invCustName").value = "";
      document.getElementById("invCustPhone").value = "";
      document.getElementById("invCustAddress").value = "";
    }
